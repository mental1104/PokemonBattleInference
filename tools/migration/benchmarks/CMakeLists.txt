# 独立benchmark只链接已验证同snapshot的正式库，不修改发布Core/CMake或重编领域实现。
cmake_minimum_required(VERSION 3.20)
project(PbiFixedBattleBenchmark LANGUAGES CXX)
find_package(Boost REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
if(NOT IS_DIRECTORY "${PBI_CORE_SOURCE_ROOT}" OR NOT IS_DIRECTORY "${PBI_CORE_ARCHIVE_DIR}")
  message(FATAL_ERROR "PBI_CORE_SOURCE_ROOT and PBI_CORE_ARCHIVE_DIR must identify the same validated snapshot")
endif()
add_library(benchmark_core STATIC IMPORTED)
set_target_properties(benchmark_core PROPERTIES IMPORTED_LOCATION "${PBI_CORE_ARCHIVE_DIR}/libpbi_core.a")
add_library(benchmark_abi STATIC IMPORTED)
set_target_properties(benchmark_abi PROPERTIES IMPORTED_LOCATION "${PBI_CORE_ARCHIVE_DIR}/libpbi_abi.a")
add_executable(pbi_fixed_benchmark fixed_battle_runner.cpp)
target_compile_features(pbi_fixed_benchmark PRIVATE cxx_std_20)
target_include_directories(pbi_fixed_benchmark PRIVATE "${PBI_CORE_SOURCE_ROOT}/include" "${PBI_CORE_SOURCE_ROOT}/src" "${PBI_CORE_SOURCE_ROOT}/../contracts/c")
target_link_libraries(pbi_fixed_benchmark PRIVATE benchmark_abi benchmark_core Boost::headers nlohmann_json::nlohmann_json)
target_compile_options(pbi_fixed_benchmark PRIVATE -Wall -Wextra -Wpedantic -Werror)
