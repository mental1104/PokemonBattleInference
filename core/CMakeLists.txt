cmake_minimum_required(VERSION 3.20)
project(PokemonBattleCore VERSION 0.2.0 LANGUAGES C CXX)

find_package(Boost REQUIRED)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
option(PBI_ENABLE_SANITIZERS "Enable address/undefined sanitizers for native validation" OFF)
if(PBI_ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(pbi_core STATIC src/calculations.cpp src/damage.cpp)
target_compile_features(pbi_core PUBLIC cxx_std_20)
target_include_directories(pbi_core PUBLIC include)
target_link_libraries(pbi_core PUBLIC Boost::headers)
set_target_properties(pbi_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

option(PBI_BUILD_ABI "Build the versioned C ABI adapter" ON)
if(PBI_BUILD_ABI)
  find_package(nlohmann_json CONFIG REQUIRED)
  add_library(pbi_abi STATIC src/abi.cpp)
  target_compile_definitions(pbi_abi PRIVATE PBI_ENGINE_VERSION="${PROJECT_VERSION}")
  target_include_directories(pbi_abi PUBLIC ../contracts/c)
  target_link_libraries(pbi_abi PUBLIC pbi_core PRIVATE nlohmann_json::nlohmann_json)
  set_target_properties(pbi_abi PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_executable(pbi_abi_vectors tests/abi_runner.cpp)
  target_link_libraries(pbi_abi_vectors PRIVATE pbi_abi)
endif()

# 这个无平台依赖的批量入口供 Python 差分验证使用，不承担 HTTP 或持久化职责。
add_executable(pbi_core_vectors tests/vector_runner.cpp)
target_link_libraries(pbi_core_vectors PRIVATE pbi_core)

include(CTest)
if(BUILD_TESTING)
  find_package(GTest CONFIG REQUIRED)
  add_executable(pbi_core_tests tests/calculations_test.cpp tests/damage_test.cpp)
  target_link_libraries(pbi_core_tests PRIVATE pbi_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(pbi_core_tests)
  if(PBI_BUILD_ABI)
    find_package(Threads REQUIRED)
    add_executable(pbi_abi_tests tests/abi_test.cpp)
    target_compile_definitions(pbi_abi_tests PRIVATE PBI_VECTOR_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../tests/vectors")
    target_link_libraries(pbi_abi_tests PRIVATE pbi_abi GTest::gtest_main nlohmann_json::nlohmann_json Threads::Threads)
    gtest_discover_tests(pbi_abi_tests)
    add_executable(pbi_c_header_test tests/c_header_test.c)
    target_link_libraries(pbi_c_header_test PRIVATE pbi_abi)
    add_test(NAME CAbi.CHeader COMMAND pbi_c_header_test)
  endif()
endif()
