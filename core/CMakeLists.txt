cmake_minimum_required(VERSION 3.20)
project(PokemonBattleCore VERSION 0.8.0 LANGUAGES C CXX)

find_package(Boost REQUIRED)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
option(PBI_ENABLE_SANITIZERS "Enable address/undefined sanitizers for native validation" OFF)
if(PBI_ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(pbi_core STATIC src/calculations.cpp src/damage.cpp src/status.cpp src/state.cpp src/profile.cpp src/actions.cpp src/graph.cpp src/transitions.cpp src/battle_graph.cpp src/turn.cpp src/configuration_stats.cpp src/configuration_moves.cpp src/configuration_space.cpp src/configuration_admission.cpp src/job_move_pool.cpp src/configuration_goals.cpp src/configuration_aggregate.cpp src/configuration_spread_search.cpp)
target_compile_features(pbi_core PUBLIC cxx_std_20)
target_include_directories(pbi_core PUBLIC include)
target_link_libraries(pbi_core PUBLIC Boost::headers)
set_target_properties(pbi_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

option(PBI_BUILD_ABI "Build the versioned C ABI adapter" ON)
if(PBI_BUILD_ABI)
  find_package(nlohmann_json CONFIG REQUIRED)
  add_library(pbi_abi STATIC src/abi.cpp src/state_contract.cpp src/graph_contract.cpp src/turn_contract.cpp src/configuration_contract.cpp src/configuration_goals_contract.cpp src/configuration_spread_search_contract.cpp)
  target_compile_definitions(pbi_abi PRIVATE PBI_ENGINE_VERSION="${PROJECT_VERSION}")
  target_include_directories(pbi_abi PUBLIC ../contracts/c)
  target_link_libraries(pbi_abi PUBLIC pbi_core PRIVATE nlohmann_json::nlohmann_json)
  set_target_properties(pbi_abi PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_executable(pbi_abi_vectors tests/abi_runner.cpp)
  target_link_libraries(pbi_abi_vectors PRIVATE pbi_abi)
  # 独立拓扑差分适配器；JSON 只存在于进程边界，纯图引擎不依赖它。
  add_executable(pbi_graph_vectors tests/graph_runner.cpp)
  target_link_libraries(pbi_graph_vectors PRIVATE pbi_core nlohmann_json::nlohmann_json)
  add_executable(pbi_battle_graph_vectors tests/battle_graph_runner.cpp)
  target_include_directories(pbi_battle_graph_vectors PRIVATE ../contracts/c)
  target_link_libraries(pbi_battle_graph_vectors PRIVATE pbi_abi nlohmann_json::nlohmann_json)
  add_executable(pbi_turn_vectors tests/turn_runner.cpp)
  target_link_libraries(pbi_turn_vectors PRIVATE pbi_abi nlohmann_json::nlohmann_json)
  # 配置差分入口只在进程边界处理 JSON；真实候选算法仍完全属于纯 Core。
  foreach(dimension stats moves space)
    add_executable(pbi_configuration_${dimension}_vectors tests/configuration_${dimension}_runner.cpp)
    target_link_libraries(pbi_configuration_${dimension}_vectors PRIVATE pbi_core nlohmann_json::nlohmann_json)
  endforeach()
  add_executable(pbi_job_move_pool_vectors tests/job_move_pool_runner.cpp)
  target_link_libraries(pbi_job_move_pool_vectors PRIVATE pbi_core nlohmann_json::nlohmann_json)
  add_executable(pbi_configuration_goals_vectors tests/configuration_goals_runner.cpp)
  target_link_libraries(pbi_configuration_goals_vectors PRIVATE pbi_abi nlohmann_json::nlohmann_json)
  add_executable(pbi_configuration_aggregate_vectors tests/configuration_aggregate_runner.cpp)
  target_link_libraries(pbi_configuration_aggregate_vectors PRIVATE pbi_core nlohmann_json::nlohmann_json)
  add_executable(pbi_configuration_spread_search_vectors tests/configuration_spread_search_runner.cpp)
  target_link_libraries(pbi_configuration_spread_search_vectors PRIVATE pbi_abi nlohmann_json::nlohmann_json)
endif()

# 这个无平台依赖的批量入口供 Python 差分验证使用，不承担 HTTP 或持久化职责。
add_executable(pbi_core_vectors tests/vector_runner.cpp)
target_link_libraries(pbi_core_vectors PRIVATE pbi_core)

include(CTest)
if(BUILD_TESTING)
  find_package(GTest CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(pbi_core_tests tests/calculations_test.cpp tests/damage_test.cpp tests/state_test.cpp tests/graph_test.cpp tests/actions_test.cpp tests/transitions_test.cpp tests/cancellation_test.cpp tests/turn_test.cpp tests/configuration_stats_test.cpp tests/configuration_moves_test.cpp tests/configuration_space_test.cpp tests/job_move_pool_test.cpp tests/configuration_goals_test.cpp tests/configuration_aggregate_test.cpp tests/configuration_spread_search_test.cpp)
  target_link_libraries(pbi_core_tests PRIVATE pbi_core GTest::gtest_main Threads::Threads)
  include(GoogleTest)
  gtest_discover_tests(pbi_core_tests)
  if(PBI_BUILD_ABI)
    find_package(Threads REQUIRED)
    add_executable(pbi_abi_tests tests/abi_test.cpp)
    target_sources(pbi_abi_tests PRIVATE tests/configuration_cancellation_test.cpp)
    target_compile_definitions(pbi_abi_tests PRIVATE PBI_VECTOR_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../tests/vectors")
    target_link_libraries(pbi_abi_tests PRIVATE pbi_abi GTest::gtest_main nlohmann_json::nlohmann_json Threads::Threads)
    gtest_discover_tests(pbi_abi_tests)
    add_executable(pbi_c_header_test tests/c_header_test.c)
    target_link_libraries(pbi_c_header_test PRIVATE pbi_abi)
    add_test(NAME CAbi.CHeader COMMAND pbi_c_header_test)
  endif()
endif()
